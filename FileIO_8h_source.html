37 FRESULT mount(FatFs& fs, BYTE pdrv = 0) override {
38 if (disk_initialize(pdrv) & STA_NOINIT) return FR_NOT_READY;
39 if (just_created) {
40 char drive_path[6];
41 snprintf(drive_path, sizeof(drive_path), "%d:", pdrv);
42 if (work_buffer == nullptr) work_buffer = new uint8_t[FF_MAX_SS];
43 fs.f_mkfs(drive_path, nullptr, work_buffer, FF_MAX_SS);
44 }
45 return IO::mount(fs, pdrv);
46 }

48 DSTATUS disk_initialize(BYTE pdrv) override {
49 if (pdrv != 0) return STA_NODISK;
50 if (file == nullptr && !open_or_create()) return STA_NODISK;
51 status = STA_CLEAR;
52 return status;
53 }
54
55 DSTATUS disk_status(BYTE pdrv) override {
56 if (pdrv != 0) return STA_NODISK;
57 return status;
58 }

60 DRESULT disk_read(BYTE pdrv, BYTE* buff, LBA_t sector, UINT count) override {
61 if (pdrv != 0) return RES_NOTRDY;
62 if (status == STA_NOINIT) return RES_NOTRDY;
63 if (seek_to(file, (uint64_t)sector * sector_size) != 0) return RES_ERROR;
64 size_t n = fread(buff, sector_size, count, file);
65 return n == count ? RES_OK : RES_ERROR;
66 }
67
68 DRESULT disk_write(BYTE pdrv, const BYTE* buff, LBA_t sector,
69 UINT count) override {
70 if (pdrv != 0) return RES_NOTRDY;
71 if (status == STA_NOINIT) return RES_NOTRDY;
72 if (seek_to(file, (uint64_t)sector * sector_size) != 0) return RES_ERROR;
73 size_t n = fwrite(buff, sector_size, count, file);
74 return n == count ? RES_OK : RES_ERROR;
75 }
76
77 DRESULT disk_ioctl(BYTE pdrv, ioctl_cmd_t cmd, void* buff) override {
78 if (pdrv != 0) return RES_PARERR;
79 switch (cmd) {
80 case CTRL_SYNC:
81 return fflush(file) == 0 ? RES_OK : RES_ERROR;
82
83 case GET_SECTOR_COUNT: {
84 if (buff == nullptr) return RES_PARERR;
85 DWORD result = (DWORD)sector_count;
86 memcpy(buff, &result, sizeof(result));
87 return RES_OK;
88 }
89
90 case GET_SECTOR_SIZE: {
91 if (buff == nullptr) return RES_PARERR;
92 WORD result = (WORD)sector_size;
93 memcpy(buff, &result, sizeof(result));
94 return RES_OK;
95 }
96
97 case GET_BLOCK_SIZE: {
98 if (buff == nullptr) return RES_PARERR;
99 DWORD result = 1;
100 memcpy(buff, &result, sizeof(result));
101 return RES_OK;
102 }
103
104 default:
105 return RES_PARERR;
106 }
107 }

119 static int seek_to(FILE* f, uint64_t pos) {
120#ifdef _WIN32
121 return _fseeki64(f, (__int64)pos, SEEK_SET);
122#else
123 return fseeko(f, (off_t)pos, SEEK_SET);
124#endif
125 }
126
127 bool open_or_create() {
128 file = fopen(path, "r+b");
129 just_created = (file == nullptr);
130 if (just_created) {
131 file = fopen(path, "w+b");
132 if (file == nullptr) return false;
133 }
134 // grow (sparsely) to the requested size if the file is smaller, whether
135 // freshly created or a pre-existing image that's too small
136 uint64_t wanted_size = (uint64_t)sector_count * sector_size;
137 if (seek_to(file, 0) != 0) return false;
138 if (fseek(file, 0, SEEK_END) != 0) return false;
139#ifdef _WIN32
140 __int64 current_size = _ftelli64(file);
141#else
142 off_t current_size = ftello(file);
143#endif
144 if (current_size < 0) return false;
145 if ((uint64_t)current_size < wanted_size) {
146 if (seek_to(file, wanted_size - 1) != 0) return false;
147 uint8_t zero = 0;
148 if (fwrite(&zero, 1, 1, file) != 1) return false;
149 fflush(file);
150 }
151 return true;
152 }
153};
