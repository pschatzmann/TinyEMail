97 bool begin(String& headers) {
98 String first;

100 headers = "";
101 if (!lines_.next(first, terminated)) {
102 return !lines_.broken(); // an empty message has no headers and one empty body
103 }
104 if (looksLikeHeader(first)) {
105 headers = first;
106 headers += "\r\n";
107 String line;
108 bool blank = false;
109 while (lines_.next(line, terminated)) {
110 if (line.length() == 0) {
111 blank = true;
112 break;
113 }
114 headers += line;
115 headers += "\r\n";
116 }

145 bool nextPart(TinyMailPart& info) {
146 if (leafPending_) {

148 info = leafInfo_;

150 return true;
151 }

155 hasPending_ = false;
156 d = pending_;
157 } else if (!scanDelimiter(d)) {

160 }
161 depth_ = 0;
162 return false;
163 }
164 depth_ = d.frame + 1;
165 if (d.closing) {
166 depth_ = d.frame; // this multipart is finished: the parent goes on
167 continue;
168 }
169 String head;
170 Delim inner;
171 bool hasInner = false;
172 readHeaders(head, inner, hasInner);
173 if (hasInner) {
174 // a delimiter inside the headers: the part is empty
175 info = TinyMailPart();
176 encoding_ = Encoding::None;
177 partOpen_ = false;
178 pending_ = inner;
179 hasPending_ = true;
180 return true;
181 }
182 String boundary;
183 if (describe(head, info, boundary)) {
184 if (boundary.length() == 0) {
185 TinyLogger::instance().warn("MIME: multipart without boundary, its parts are skipped");
186 continue;
187 }
188 if (depth_ >= maxDepth) {
189 TinyLogger::instance().warn("MIME: multipart nested too deep, its parts are skipped");
190 continue;
191 }
192 boundaries_[depth_++] = boundary;
193 continue; // the parts of the new multipart come next
194 }
195 startBody();

197 }
198 return false;

328 for (int i = depth_ - 1; i >= 0; i--) {

368 bool describe(const String& headers, TinyMailPart& info, String& boundary) {
369 String contentType = TinyMail::headerValue(headers, "Content-Type");
370 String type = TinyText::lowerCopy(contentType);
371 int semicolon = type.indexOf(';');
372 if (semicolon >= 0) {
373 type = type.substring(0, semicolon);
374 }
375 type.trim();
376 if (type.length() == 0) {
377 type = "text/plain"; // default when the header is missing
378 }
379 if (type.startsWith("multipart/")) {
380 boundary = paramValue(contentType, "boundary");
381 return true;
382 }
383
384 String disposition = TinyMail::headerValue(headers, "Content-Disposition");
385 info.contentType = type;
386 info.attachment = TinyText::lowerCopy(disposition).startsWith("attachment");
387 info.filename = fileName(disposition, contentType);
388
389 String encodingText = TinyText::lowerCopy(TinyMail::headerValue(headers, "Content-Transfer-Encoding"));
390 encodingText.trim();
391 encoding_ = Encoding::None;
392 if (encodingText == "base64") {
393 encoding_ = Encoding::Base64;
394 } else if (encodingText == "quoted-printable") {
395 encoding_ = Encoding::QuotedPrintable;
396 } else if (encodingText.length() > 0 && encodingText != "7bit" && encodingText != "8bit" &&
397 encodingText != "binary") {
398 TinyLogger::instance().warn(String("MIME: transfer encoding not supported, data passed as is: ") +
399 encodingText);
400 }
401 return false;
402 }

405 static String fileName(const String& disposition, const String& contentType) {
406 String name = paramValue(disposition, "filename*");
407 if (name.length() > 0) {
408 // RFC 2231: charset'language'percent-encoded value
409 int q1 = name.indexOf('\'');
410 int q2 = q1 >= 0 ? name.indexOf('\'', q1 + 1) : -1;
411 if (q1 >= 0 && q2 >= 0) {
412 return TinyText::toUtf8(name.substring(0, q1), TinyText::percentDecode(name.substring(q2 + 1)));
413 }
414 return TinyText::percentDecode(name);
415 }
416 name = paramValue(disposition, "filename");
417 if (name.length() == 0) {
418 name = paramValue(contentType, "name");
419 }
420 return TinyText::decodeWords(name);
421 }

424 static String paramValue(const String& value, const char* param) {
425 String lower = TinyText::lowerCopy(value);
426 String key = param;
427 key += '=';
428 int start = -1;
429 int from = 0;
430 while (true) {
431 int k = lower.indexOf(key, from);
432 if (k < 0) {
433 break;
434 }
435 // a parameter starts after ';' or a space: "name=" is not part of "filename="
436 if (k == 0 || lower.charAt(k - 1) == ';' || lower.charAt(k - 1) == ' ' ||
437 lower.charAt(k - 1) == '\t') {
438 start = k;
439 break;
440 }
441 from = k + 1;
442 }
443 if (start < 0) {
444 return "";
445 }
446 start += key.length();
447 if (start < static_cast<int>(value.length()) && value.charAt(start) == '"') {
448 int close = value.indexOf('"', start + 1);
449 if (close < 0) {
450 close = value.length();
451 }
452 return value.substring(start + 1, close);
453 }
454 int end = value.indexOf(';', start);
455 if (end < 0) {
456 end = value.length();
457 }
458 String out = value.substring(start, end);
459 out.trim();
460 return out;
461 }

582 int nextBase64() {
583 while (outPos_ >= outLen_) {
584 uint32_t value = 0;
585 int n = 0;
586 bool eof = false;
587 while (n < 4) {
588 int c = raw();
589 if (c < 0) {
590 eof = true;
591 break;
592 }
593 if (c == '=') {
594 if (n == 0) {
595 continue; // padding of an earlier group
596 }
597 break; // padding ends this group
598 }
599 int d = TinyText::base64Value(static_cast<char>(c));
600 if (d >= 0) {
601 value = (value << 6) | static_cast<uint32_t>(d);
602 n++;
603 }
604 }
605 if (n < 2) {
606 if (eof) {
607 return -1;
608 }
609 continue;
610 }
611 value <<= 6 * (4 - n);
612 out_[0] = static_cast<uint8_t>(value >> 16);
613 out_[1] = static_cast<uint8_t>(value >> 8);
614 out_[2] = static_cast<uint8_t>(value);
615 outLen_ = static_cast<uint8_t>(n - 1);
616 outPos_ = 0;
617 }
618 return out_[outPos_++];
619 }

635 int nextQuotedPrintable() {
636 for (;;) {
637 if (queuePos_ < queue_.length()) {
638 return static_cast<uint8_t>(queue_.charAt(queuePos_++));
639 }
640 queue_ = "";
641 queuePos_ = 0;
642 int c = pull();
643 if (c < 0) {
644 whitespace_ = ""; // whitespace at the end of the data is removed
645 return -1;
646 }
647 if (c == ' ' || c == '\t') {
648 whitespace_ += static_cast<char>(c);
649 continue;
650 }
651 if (c == '\r' || c == '\n') {
652 whitespace_ = ""; // whitespace before a line break is removed (RFC 2045)
653 return c;
654 }
655 if (whitespace_.length() > 0) {
656 // a visible byte follows: the whitespace is part of the data
657 push(c);
658 queue_ = whitespace_;
659 whitespace_ = "";
660 continue;
661 }
662 if (c != '=') {
663 return c;
664 }
665 int a = pull();
666 if (a < 0) {
667 return '=';
668 }
669 if (a == '\n') {
670 continue; // soft line break
671 }
672 if (a == '\r') {
673 int b = pull();
674 if (b == '\n') {
675 continue; // soft line break
676 }
677 push(b);
678 push(a);
679 return '=';
680 }
681 int b = pull();
682 if (TinyMailData::hexValue(static_cast<char>(a)) >= 0 &&
683 TinyMailData::hexValue(static_cast<char>(b)) >= 0) {
684 return TinyMailData::hexValue(static_cast<char>(a)) * 16 +
685 TinyMailData::hexValue(static_cast<char>(b));
686 }
687 push(b);
688 push(a);
689 return '='; // not an escape: keep it
690 }
691 }

693 static int hexValue(char c) {
694 if (c >= '0' && c <= '9') return c - '0';
695 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
696 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
697 return -1;
