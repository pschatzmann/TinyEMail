144 static String headerValue(const String& block, const char* name) {
145 size_t nameLen = strlen(name);
146 String value;
147 bool found = false;
148 int pos = 0;
149 while (pos < static_cast<int>(block.length())) {
150 int end = block.indexOf('\n', pos);
151 if (end < 0) {
152 end = block.length();
153 }
154 String line = block.substring(pos, end);
155 if (line.endsWith("\r")) {
156 line.remove(line.length() - 1);
157 }
158 bool continuation =
159 line.length() > 0 && (line.charAt(0) == ' ' || line.charAt(0) == '\t');
160 if (continuation && found) {
161 line.trim();
162 value += ' ';
163 value += line;
164 } else if (found) {
165 break; // the header ended
166 } else if (line.length() > nameLen && line.charAt(nameLen) == ':' &&
167 sameName(line, name, nameLen)) {
168 found = true;
169 value = line.substring(nameLen + 1);
170 value.trim();
171 }
172 pos = end + 1;
173 }
174 return value;
175 }

178 static bool sameName(const String& line, const char* name, size_t n) {
179 for (size_t i = 0; i < n; i++) {
180 char a = line.charAt(i);
181 char b = name[i];
182 if (a >= 'A' && a <= 'Z') a = a + 32;
183 if (b >= 'A' && b <= 'Z') b = b + 32;
184 if (a != b) {
185 return false;
186 }
187 }
188 return true;
189 }
