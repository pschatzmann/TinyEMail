138 int readReply() {
139 lastCode_ = -1;
140 lastReply_ = "";
141 while (true) {
142 String line;
143 bool complete = TinyText::readLine(*this, line, maxReplyLength);
144 if (!complete) {
145 broken_ = true; // timeout, or a line longer than maxReplyLength
146 TinyLogger::instance().error("SMTP: no complete reply from the server (timeout or line too long)");
147 return -1;
148 }
149 line.trim();
150 if (line.length() < 3 || !isDigitChar(line.charAt(0)) ||
151 !isDigitChar(line.charAt(1)) || !isDigitChar(line.charAt(2))) {
152 lastReply_ = line;
153 TinyLogger::instance().error(String("SMTP: invalid reply: ") + line);
154 return -1;
155 }
156 TinyLogger::instance().debug(String("S: ") + line);
157 if (lastReply_.length() > 0) {
158 lastReply_ += '\n';
159 }
160 lastReply_ += line;
161 lastCode_ = (line.charAt(0) - '0') * 100 + (line.charAt(1) - '0') * 10 +
162 (line.charAt(2) - '0');
163 // "250 text" is the last line, "250-text" is followed by more lines
164 if (line.length() == 3 || line.charAt(3) == ' ') {
165 break;
166 }
167 if (line.charAt(3) != '-') {
168 TinyLogger::instance().error(String("SMTP: invalid reply: ") + line);
169 lastCode_ = -1;
170 return -1;
171 }
172 }
173 return lastCode_;
174 }

196 bool authPlain(const char* user, const char* appPassword) {
197 if (user == nullptr || appPassword == nullptr) {
198 TinyLogger::instance().error("SMTP: AUTH PLAIN without user or password");
199 return false;
200 }
201 size_t userLen = strlen(user);
202 size_t passLen = strlen(appPassword);
203 if (userLen + passLen + 2 > maxCredentialsSize) {
204 TinyLogger::instance().error("SMTP: user and password are too long for AUTH PLAIN");
205 return false;
206 }
207 // SASL PLAIN: authzid (empty) NUL authcid NUL passwd
208 uint8_t raw[maxCredentialsSize];
209 size_t n = 0;
210 raw[n++] = 0;
211 memcpy(raw + n, user, userLen);
212 n += userLen;
213 raw[n++] = 0;
214 memcpy(raw + n, appPassword, passLen);
215 n += passLen;
216
217 char encoded[((maxCredentialsSize + 2) / 3) * 4 + 1];
218 size_t len = TinyText::base64Encode(raw, n, encoded);
219 encoded[len] = '\0';
220
221 TinyLogger::instance().debug("C: AUTH PLAIN ****"); // the credentials are never logged
222 print("AUTH PLAIN ");
223 print(encoded);
224 print("\r\n");
225 flush();
226 return expect(235);
227 }

423 static bool isValidMail(const TinyMail& mail) {
424 if (mail.to.size() + mail.bcc.size() == 0) {
425 return false; // at least one recipient
426 }
427 for (const String& address : mail.to) {
428 if (!isAddress(address.c_str())) {
429 return false;
430 }
431 }
432 for (const String& address : mail.bcc) {
433 if (!isAddress(address.c_str())) {
434 return false;
435 }
436 }
437 for (size_t i = 0; i < mail.attachments.size(); i++) {
438 const TinyMailAttachment& a = mail.attachments[i];
439 // the content type is a header value: no line breaks, parameters or quotes
440 if (!isSafe(a.contentType) || strpbrk(a.contentType, "; \\\"") != nullptr ||
441 a.stream == nullptr) {
442 return false;
443 }
444 if (i == 0) {
445 if (strncmp(a.contentType, "text/", 5) != 0) {
446 return false; // the text body
447 }
448 continue;
449 }
450 if (a.filename == nullptr || !isSafe(a.filename) ||
451 strpbrk(a.filename, "\"\\") != nullptr) {
452 return false;
453 }
454 }

530 void dataChar(char c) {
531 if (pendingCR_) {
532 pendingCR_ = false;
533 print("\r\n"); // the held back CR is a line break
534 dataLineStart_ = true;
535 if (c == '\n') {
536 return; // this LF completes that CRLF
537 }
538 }
539 if (c == '\r') {
540 pendingCR_ = true; // wait for the LF: CRLF is written once
541 return;
542 }
543 if (c == '\n') {
544 print("\r\n");
545 dataLineStart_ = true;
546 return;
547 }
548 if (dataLineStart_ && c == '.') {
549 write(static_cast<uint8_t>('.')); // dot stuffing
550 }
551 write(static_cast<uint8_t>(c));
552 dataLineStart_ = false;
553 }

585 void writeBody(const TinyMailAttachment& a) {

591 qpByte(static_cast<uint8_t>(c));

601 void qpByte(uint8_t b) {
602 if (qpCR_) {
603 qpCR_ = false;
604 qpHardBreak();
605 if (b == '\n') {
606 return; // this LF completes the CRLF
607 }
608 }
609 if (b == '\r') {
610 qpCR_ = true; // wait for the LF
611 return;
612 }
613 if (b == '\n') {
614 qpHardBreak();
615 return;
616 }
617 if (b == ' ' || b == '\t') {
618 if (qpPendingCount_ < qpPendingMax) {
619 qpPending_[qpPendingCount_++] = b; // written only if no line break follows
620 } else {
621 qpEncodePending();
622 qpEncodeByte(b);
623 }
624 return;
625 }
626 qpFlushPending(); // a space before a visible character is written as it is
627 if (b >= 33 && b <= 126 && b != '=') {
628 char c = static_cast<char>(b);
629 qpToken(&c, 1);
630 } else {
631 qpEncodeByte(b);
632 }
633 }
634
636 void qpHardBreak() {
637 qpEncodePending();
638 dataChar('\r');
639 dataChar('\n');

641 }

643 void qpFlushPending() {

645 char c = static_cast<char>(qpPending_[i]);
646 qpToken(&c, 1);
647 }
648 qpPendingCount_ = 0;
649 }

651 void qpEncodePending() {
652 for (uint8_t i = 0; i < qpPendingCount_; i++) {
653 qpEncodeByte(qpPending_[i]);
654 }
655 qpPendingCount_ = 0;
656 }

658 void qpEncodeByte(uint8_t b) {
659 static const char hex[] = "0123456789ABCDEF";
660 char token[3] = {'=', hex[b >> 4], hex[b & 0x0F]};

697 void base64Put(uint8_t b) {

706 void base64Quad(const uint8_t* group, uint8_t n) {
707 static const char alphabet[] =
708 "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
709 uint32_t triple = static_cast<uint32_t>(group[0]) << 16;
710 if (n > 1) {
711 triple |= static_cast<uint32_t>(group[1]) << 8;
712 }
713 if (n > 2) {
714 triple |= group[2];
715 }
716 char out[4] = {alphabet[(triple >> 18) & 0x3F], alphabet[(triple >> 12) & 0x3F],
717 n > 1 ? alphabet[(triple >> 6) & 0x3F] : '=',
718 n > 2 ? alphabet[triple & 0x3F] : '='};
719 for (int i = 0; i < 4; i++) {
720 write(static_cast<uint8_t>(out[i]));
721 }
722 b64Column_ += 4;
723 if (b64Column_ >= 76) {
724 print("\r\n");
725 b64Column_ = 0;
726 }
727 }
