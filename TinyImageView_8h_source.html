37 bool load(fatfs::SDClass& fs, const String& path) {
38 loaded_ = false;
39 error_ = "";
40 fatfs::File f = fs.open(path.c_str(), FILE_READ);
41 if (!f) {
42 error_ = "Cannot open the picture";
43 return false;
44 }
45 uint8_t head[8] = {0};
46 f.read(head, sizeof(head));
47 f.seek(0);
48 const size_t size = f.available();
49
50 if (head[0] == 0xFF && head[1] == 0xD8) {
51 tinygpu::JPEGParser<RGB565> parser(sprite_);
52 loaded_ = parser.decode(f); // streamed from the file
53 } else if (size > kMaxBufferedBytes) {
54 error_ = "Picture too large to show";
55 } else if (memcmp(head, "\x89PNG", 4) == 0) {
56 loaded_ = decodeBuffered<tinygpu::PNGParser<RGB565>>(f, size);
57 } else if (memcmp(head, "GIF8", 4) == 0) {
58 loaded_ = decodeBuffered<tinygpu::GIFParser<RGB565>>(f, size);
59 } else if (head[0] == 'B' && head[1] == 'M') {
60 loaded_ = decodeBmp(f, size);
61 } else {
62 error_ = "Picture format not supported";
63 }
64 f.close();
65 if (!loaded_ && error_.length() == 0) error_ = "Cannot decode the picture";
66 return loaded_;
67 }

96 bool decodeBuffered(fatfs::File& f, size_t size) {
97 uint8_t* data = new uint8_t[size];
98 const size_t n = f.read(data, size);
99 Parser parser(sprite_);

108 const size_t n = f.read(data, size);
109 tinygpu::BMPParser<RGB565> parser(sprite_);
110 parser.write(data, n);
111 const bool ok = parser.isComplete();
112 delete[] data;
113 if (!ok) error_ = "Cannot decode the picture";
