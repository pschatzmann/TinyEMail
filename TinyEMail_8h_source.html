118 bool login(const char* user, const char* appPassword) {
119 if (!TinySMTP::isAddress(user)) {
120 TinyLogger::instance().error("login: the user must be an e-mail address without spaces");
121 forget();
122 return false;
123 }
124 if (!TinySMTP::isSafe(appPassword)) {
125 TinyLogger::instance().error("login: the password is empty or has a line break");
126 forget();
127 return false;
128 }
129 TinySMTP smtp(*client_);
130 smtp.setTimeout(timeoutMs_);
131 bool ok = smtp.begin(smtpHost_, smtpPort_) &&
132 check(smtp.readReply() == 220, smtp, "login: greeting") &&
133 check(smtp.ehlo(), smtp, "login: EHLO") &&
134 check(smtp.authPlain(user, appPassword), smtp, "login: AUTH PLAIN");
135 if (ok && !smtp.quit()) {
136 TinyLogger::instance().warn("login: the server did not confirm QUIT");
137 }
138 remember(smtp);
139 smtp.end();
140 if (!ok) {
141 forget();
142 TinyLogger::instance().error("login failed, the account is not logged in");
143 return false;
144 }
145 user_ = user;
146 password_ = appPassword;
147 TinyLogger::instance().info("login accepted");
148 return true;
149 }

176 check(smtp.ehlo(), smtp, "send: EHLO") &&

178 check(smtp.mailFrom(user_.c_str()), smtp, "send: MAIL FROM") &&

183 if (ok && !smtp.quit()) {
184 TinyLogger::instance().warn("send: the server did not confirm QUIT, the mail was accepted");
185 }
186 remember(smtp);
187 smtp.end();
188 if (ok) {
189 TinyLogger::instance().info("send: mail accepted by the server");
190 }

283 template <typename Callback>
284 int receive(Callback callback, Filter filter = Filter::Unread) {
285 if (!loggedIn()) {
286 TinyLogger::instance().error("receive: not logged in, call login() first");
287 return -1;
288 }
289
290 TinyIMAP imap(*client_);
291 imap.setTimeout(timeoutMs_);
292 imap.setLimits(32768, maxMessageSize_);
293 bool session = imap.begin(imapHost_, imapPort_) && imap.readGreeting();
294 if (session && !imap.login(user_.c_str(), password_.c_str())) {
295 TinyLogger::instance().error(String("receive: login failed: ") + reason(imap));
296 session = false;
297 }
298 if (session && !imap.select("INBOX")) {
299 TinyLogger::instance().error(String("receive: cannot open INBOX: ") + reason(imap));
300 session = false;
301 }
302
303 int count = -1;
304 if (session) {
305 String uids;
306 if (!imap.searchList(searchKey(filter), uids)) {
307 TinyLogger::instance().error(String("receive: search failed: ") + reason(imap));
308 } else {
309 count = 0;
310 Receiver<Callback> receiver(callback);
311 const char* p = uids.c_str();
312 while (receiver.keepGoing) {
313 char* end = nullptr;
314 unsigned long uid = strtoul(p, &end, 10);
315 if (end == p) {
316 break; // no more UIDs
317 }
318 p = end;
319 receiver.uid = static_cast<uint32_t>(uid);
320 receiver.flag = Mail::FlagChange::None;
321 if (!imap.fetchStreamed(receiver.uid, "BODY.PEEK[]", receiver)) {
322 if (imap.broken() || !imap.connected()) {
323 TinyLogger::instance().error(String("receive: connection lost after ") +
324 String(receiver.count) + " mails");
325 break;
326 }
327 TinyLogger::instance().warn(String("receive: mail ") + String(static_cast<unsigned long>(uid)) +
328 " could not be read, skipped");
329 }
330 // the message has been read to its end: now the requested flag can be sent
331 if (receiver.flag != Mail::FlagChange::None && imap.connected() && !imap.broken()) {
332 bool ok = true;
333 switch (receiver.flag) {
334 case Mail::FlagChange::Read:
335 ok = imap.setFlag(receiver.uid, "\\Seen", true);
336 break;
337 case Mail::FlagChange::Unread:
338 ok = imap.setFlag(receiver.uid, "\\Seen", false);
339 break;
340 case Mail::FlagChange::Flagged:
341 ok = imap.setFlag(receiver.uid, "\\Flagged", true);
342 break;
343 case Mail::FlagChange::Unflagged:
344 ok = imap.setFlag(receiver.uid, "\\Flagged", false);
345 break;
346 default:
347 break;
348 }
349 if (!ok) {
350 TinyLogger::instance().warn("receive: the flag of a mail could not be set on the server");
351 }
352 }
353 }
354 count = receiver.count;
355 }
356 }
357 lastCode_ = -1;
358 lastReply_ = imap.lastResponse();
359 if (session && imap.connected() && !imap.broken() && !imap.logout()) {
360 TinyLogger::instance().warn("receive: the server did not confirm LOGOUT");
361 }
362 imap.end();
363 if (count >= 0) {
364 TinyLogger::instance().info(String("receive: ") + String(count) + " mails read");
365 }
366 return count;
367 }

449 bool handle(Stream& raw, size_t) override {
450 TinyMIMEReader reader(raw, mimeLineLength);
451 Mail mail;
452 mail.uid = uid;
453 if (!reader.begin(mail.headers)) {
454 TinyLogger::instance().warn("receive: the message cannot be read, skipped");
455 return false;
456 }
457 mail.from = mail.header("From");
458 mail.subject = mail.header("Subject");
459 count++;
460 mail.reader_ = &reader; // the parts are read from the stream while the callback runs
461 keepGoing = callback_(static_cast<const Mail&>(mail));
462 flag = mail.flagChange();
463 mail.reader_ = nullptr;
464 return !reader.broken();
465 }
