61inline void drawMail(tinygpu::ISurface<RGB565>& target, const Bounds& r, RGB565 color, uint8_t thickness) {
62 const int32_t x0 = r.x + 2;
63 const int32_t x1 = r.x + r.w - 3;
64 const int32_t y0 = r.y + r.h / 5;
65 const int32_t y1 = r.y + r.h - r.h / 5 - 1;
66 const int32_t xm = (x0 + x1) / 2;
67 const int32_t ym = y0 + (y1 - y0) * 6 / 10;
68 drawLineThick<RGB565>(target, x0, y0, x1, y0, color, thickness);
69 drawLineThick<RGB565>(target, x1, y0, x1, y1, color, thickness);
70 drawLineThick<RGB565>(target, x1, y1, x0, y1, color, thickness);
71 drawLineThick<RGB565>(target, x0, y1, x0, y0, color, thickness);
72 drawLineThick<RGB565>(target, x0, y0, xm, ym, color, thickness); // the flap
73 drawLineThick<RGB565>(target, xm, ym, x1, y0, color, thickness);

399 static String withoutOffset(const String& date) {
400 String out = date;
401 out.trim();
402 const int n = out.length();
403 if (n >= 6 && (out.charAt(n - 5) == '+' || out.charAt(n - 5) == '-')) {
404 out = out.substring(0, n - 5);
405 out.trim();
406 }
407 return out;

452 // An attachment: a text file is shown as text, any other file shows its name and size
453 void showAttachment(int k) {
454 currentAttachment_ = k;
455 view_ = View::kAttachment;
456 showMailColors();
457 String name = store_.attachmentName(currentMail_, k);
458 String path = store_.attachmentPath(currentMail_, k);
459 int size = 0;
460 fatfs::File f = store_.fs().open(path.c_str(), FILE_READ);
461 if (f) {
462 size = f.available();
463 f.close();
464 }
465 bool html = endsWith(name, ".html") || endsWith(name, ".htm");
466 bool text = html || endsWith(name, ".txt") || endsWith(name, ".csv") ||
467 endsWith(name, ".log") || endsWith(name, ".md") || endsWith(name, ".json") ||
468 endsWith(name, ".xml");
469 // the labels "Name:" and "Size:" are in the small font (byte 0x01 switches it)
470 String header = String("\x01") + "Name:" + "\x01" + " " + name + "\n" + String("\x01") + "Size:" + "\x01" +
471 " " + String(size) + " bytes\n";
472 attachmentCount_ = 0; // no attachment badges in this view
473 if (TinyImageView::isPicture(name)) {
474 imageView_.load(store_.fs(), path); // shows the error text if it cannot be decoded
475 layoutMail(header, "", true, true);
476 } else {
477 if (!text) {
478 header += "This file is not text. It is stored in the mail folder on the storage.\n";
479 }
480 layoutMail(header, text ? path : String(""), !html);
481 }
482 appBar_.leading = &backButton_;
483 appBar_.setTitle(name.c_str());
484 listView_.visible = false;
485 mailView_.visible = true;
486 app_.screen().invalidate();
487 }

495 // Lays out the mail view top to bottom: header, divider, body, then the
496 // attachment badges (wrapped in rows). A badge is as wide as its name plus padding.
497 void layoutMail(const String& header, const String& path, bool plain, bool picture = false) {
498 mailHeader_.bounds = Bounds(0, kBarH, kTextW, 0);
499 mailHeader_.setSource(header, store_.fs(), "", true);
500 int32_t y = mailHeader_.bounds.y + mailHeader_.bounds.h;
501
502 if (picture) {
503 bodyWidget_ = &imageView_;
504 imageView_.bounds = Bounds(0, y, kTextW, 0);
505 imageView_.layout(); // sets the height of the picture
506 } else {
507 bodyWidget_ = &mailBody_;
508 mailBody_.bounds = Bounds(0, y, kTextW, 0);
509 mailBody_.setSource("", store_.fs(), path, plain); // sets the height of the body
510 }
511 y = bodyWidget_->bounds.y + bodyWidget_->bounds.h + 12;
512
513 const int32_t maxX = kTextW - 12;
514 int32_t x = 12;
515 int32_t rowY = y;
516 for (int k = 0; k < attachmentCount_; k++) {
517 const int32_t w = static_cast<int32_t>(chipFont().measureTextWidth(attachmentNames_[k].c_str())) + 24;
518 if (x > 12 && x + w > maxX) {
519 x = 12;
520 rowY += kChipH + 6;
521 }
522 attachmentChips_[k].bounds = Bounds(x, rowY, w, kChipH);
523 x += w + 6;
524 }
525 }

700 // The setup file has one setting per line: dark=1 or dark=0, address=..., password=...
701 // Without a line for the account, the account of the configuration is used.
702 void loadSetup() {
703 fatfs::File f = store_.fs().open(kSetupFile, FILE_READ);
704 if (!f) return;
705 while (f.available()) {
706 String line = f.readStringUntil('\n');
707 line.trim();
708 if (line == "dark=1") {
709 darkMode_ = true;
710 } else if (line == "dark=0") {
711 darkMode_ = false;
712 } else if (line.startsWith("address=")) {
713 userAddress_ = line.substring(strlen("address=")).c_str();
714 } else if (line.startsWith("password=")) {
715 userPassword_ = line.substring(strlen("password=")).c_str();
716 } else if (line.startsWith("imap=")) {
717 imapHost_ = line.substring(strlen("imap=")).c_str();
718 } else if (line.startsWith("port=")) {
719 imapPort_ = static_cast<uint16_t>(atoi(line.substring(strlen("port=")).c_str()));
720 } else if (line.startsWith("smtp=")) {
721 smtpHost_ = line.substring(strlen("smtp=")).c_str();
722 } else if (line.startsWith("smtpport=")) {
723 smtpPort_ = static_cast<uint16_t>(atoi(line.substring(strlen("smtpport=")).c_str()));
724 }
725 }
726 f.close();
727 }

788 // "a@x.com, b@y.com; c@z.com" becomes three addresses: the separators are ',' and ';'
789 static std::vector<String> splitAddresses(const std::string& list) {
790 std::vector<String> out;
791 String current;
792 for (size_t i = 0; i <= list.size(); i++) {
793 const char c = i < list.size() ? list[i] : ',';
794 if (c == ',' || c == ';') {
795 current.trim();
796 if (current.length() > 0) out.push_back(current);
797 current = "";
798 } else {
799 current += c;
800 }
801 }
802 return out;
803 }

805 // Sends the mail over SMTP and keeps a copy in the Sent folder. The title shows what went wrong.
806 void sendMail() {
807 keyboard_.hide();
808 const std::string to = toField_.text();
809 const std::string subject = subjectField_.text();
810 const std::string body = bodyField_.text();
811 const std::vector<String> recipients = splitAddresses(to);
812 if (recipients.empty()) {
813 appBar_.setTitle("Enter a recipient");
814 return;
815 }
816 if (!connectWifi()) return; // the WiFi page is shown; send again once connected
817 WiFiClientSecure secure;
818 secure.setInsecure(); // certificate checking is not implemented yet
819 TinyEMail mail(secure);
820 mail.setSmtpServer(smtpHost_.c_str(), smtpPort_);
821 if (!mail.login(userAddress_.c_str(), userPassword_.c_str())) {
822 appBar_.setTitle("Login failed: not sent");
823 return;
824 }
825 // one mail to all the recipients; the body is its first (text) attachment
826 TinyTextStream bodyText(body.c_str());
827 TinyEMail::Attachment text;
828 text.contentType = "text/plain";
829 text.stream = &bodyText;
830 TinyEMail::Mail outgoing;
831 for (const String& address : recipients) {
832 outgoing.to.push_back(address);
833 }
834 outgoing.subject = subject.c_str();
835 outgoing.attachments.push_back(text);
836 if (!mail.send(outgoing)) {
837 appBar_.setTitle("Not sent");
838 return;
839 }
840 const String uid = String(static_cast<unsigned long>(millis()));
841 store_.add(uid, String("Me <") + userAddress_.c_str() + ">", subject.c_str(), body.c_str(), "", "", "", "",
842 "sent");
843 composeMode_ = false;
844 signInButton_.setLabel("Sign in");
845 setFilter(Filter::kSent);
846 showList();
847 }

867 FetchStatus fetchMail(const char* address, const char* password) {
868 if (!connectWifi()) return FetchStatus::kNoWifi;
869 WiFiClientSecure secure;

871 TinyEMail mail(secure);

873 mail.setSmtpServer(smtpHost_.c_str(), smtpPort_);
