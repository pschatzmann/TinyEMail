99 bool save(const TinyEMail::Mail& mail) {
100 String name = "m" + mail.uid;
101 if (indexOf(name) >= 0) return true;
102 if (count_ >= kMaxMails) {

111 String attachmentLines;
112 bool htmlWritten = false;
113 bool textWritten = false;
114 int partNo = 0;
115 for (TinyPart& part : mail.parts()) {
116 partNo++;

189 bool add(const String& uid, const String& from, const String& subject, const String& text,

191 const String& attachmentData = "", const String& date = "", const String& box = "inbox") {

195 if (!fs_.exists(dir.c_str()) && !fs_.mkdir(dir.c_str())) return false;

207
208 folders_[count_] = name;
209 froms_[count_] = from;
210 subjects_[count_] = subject;
211 count_++;
212 return true;

219 bool attachFile(const String& uid, const String& fileName, const uint8_t* data, size_t size) {
220 int i = indexOf("m" + uid);
221 if (i < 0) return false;
222 String dir = String(kRoot) + "/" + folders_[i];
223 String name = sanitize(fileName);
224 String lines = "From: " + froms_[i] + "\nSubject: " + subjects_[i] + "\nDate: " + dates_[i] + "\nBox: " +
225 boxes_[i] + "\n";
226 int n = attachmentCount(i);
227 for (int k = 0; k < n; k++) {
228 lines += String(kAttachment) + attachmentName(i, k) + "\n";
229 }
230 lines += String(kAttachment) + name + "\n";
231
232 fs_.remove((dir + "/" + name).c_str());
233 fatfs::File f = fs_.open((dir + "/" + name).c_str(), FILE_WRITE);
234 if (!f) return false;
235 f.write(data, size);
236 f.close();
237 return writeText(dir + "/mail.txt", lines);

272 void loadIndex() {
273 count_ = 0;
274 fatfs::directory_iterator it = fs_.directoryIterator(kRoot);
275 while (it != fatfs::directory_iterator::end() && count_ < kMaxMails) {
276 fatfs::directory_entry entry = *it;
277 ++it;
278 if (!entry.is_directory) continue;
279 String full(entry.path.c_str());
280 String name = full.substring(full.lastIndexOf('/') + 1);
281 if (!name.startsWith("m")) continue;
282 folders_[count_++] = name;
283 }
284 // read the headers after the scan, one file at a time
285 for (int i = 0; i < count_; i++) {
286 String mailFile = String(kRoot) + "/" + folders_[i] + "/mail.txt";
287 froms_[i] = headerValue(mailFile, "From: ");
288 subjects_[i] = headerValue(mailFile, "Subject: ");
289 dates_[i] = headerValue(mailFile, "Date: ");
290 boxes_[i] = headerValue(mailFile, "Box: ");
291 if (boxes_[i].length() == 0) boxes_[i] = "inbox";
292 }
293 }

314 bool copyToFile(Stream& in, const String& filePath) {
315 fs_.remove(filePath.c_str());
316 fatfs::File out = fs_.open(filePath.c_str(), FILE_WRITE);
317 if (!out) return false;
318 uint8_t buf[256];
319 size_t n = 0;
320 int c;
321 while ((c = in.read()) >= 0) {
322 buf[n++] = static_cast<uint8_t>(c);
323 if (n == sizeof(buf)) {
324 out.write(buf, n);
325 n = 0;
326 }
327 }
328 if (n > 0) out.write(buf, n);
329 out.close();
330 return true;
331 }

334 static String sanitize(const String& name) {
335 String out;
336 for (unsigned int i = 0; i < name.length() && out.length() < 40; i++) {
337 char c = name.charAt(i);
338 bool ok = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') ||
339 c == '.' || c == '-' || c == '_';
340 out += ok ? c : '_';
341 }
342 if (out.length() == 0) out = "attachment";
343 return out;
344 }
