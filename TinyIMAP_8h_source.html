393 static bool nextLine(const String& text, int& pos, String& line) {
394 if (pos < 0 || pos >= static_cast<int>(text.length())) {
395 return false;
396 }
397 int end = text.indexOf('\n', pos);
398 if (end < 0) {
399 end = text.length();
400 }
401 line = text.substring(pos, end);
402 pos = end + 1;
403 return true;

407 static int literalSizeAt(const String& line) {
408 if (!line.endsWith("}")) {
409 return -1;
410 }
411 int open = line.lastIndexOf('{');
412 if (open < 0 || open >= static_cast<int>(line.length()) - 2) {
413 return -1;
414 }
415 const char* digits = line.c_str() + open + 1;
416 char* end = nullptr;
417 unsigned long size = strtoul(digits, &end, 10);
418 if (end != line.c_str() + line.length() - 1 || end == digits) {
419 return -1;
420 }
421 return static_cast<int>(size);

426 bool run(const char* text, String* literal = nullptr, LiteralHandler* handler = nullptr) {
427 if (text == nullptr || strpbrk(text, "\r\n") != nullptr) {
428 TinyLogger::instance().error("IMAP: command with a line break refused");
429 return false; // no line breaks: they would inject commands
430 }
431 lastUntagged_ = "";
432 lastResponse_ = "";
433 literalSize_ = -1;
434
435 tagCounter_ = (tagCounter_ % 999) + 1;
436 snprintf(tag_, sizeof(tag_), "A%03u", static_cast<unsigned>(tagCounter_));
437 size_t tagLen = strlen(tag_);
438
439 // the password of LOGIN is never logged
440 bool secret = strncmp(text, "LOGIN ", 6) == 0;
441 TinyLogger::instance().debug(String("C: ") + tag_ + " " + (secret ? "LOGIN ****" : text));
442 print(tag_);
443 print(' ');
444 print(text);
445 print("\r\n");
446 flush();
447
448 bool dataOk = true; // false if the handler did not accept the literal
449 while (true) {
450 String line;
451 if (!readLine(line)) {
452 return false; // timeout or too long
453 }
454 if (line.startsWith(tag_) && line.charAt(tagLen) == ' ') {
455 lastResponse_ = line;
456 bool ok = strncmp(line.c_str() + tagLen + 1, "OK", 2) == 0 && dataOk;
457 if (!ok) {
458 TinyLogger::instance().warn(String("IMAP: command failed: ") + line);
459 }
460 return ok;
461 }
462 TinyLogger::instance().debug(String("S: ") + line);
463 if (lastUntagged_.length() + line.length() > maxLineLength_ * 4) {
464 broken_ = true;
465 TinyLogger::instance().error("IMAP: responses too long, connection dropped");
466 return false;
467 }
468 lastUntagged_ += line;
469 lastUntagged_ += '\n';
470
471 int size = literalSizeAt(line);
472 if (size >= 0) {
473 if (handler != nullptr) {
474 if (!readHandled(static_cast<size_t>(size), *handler)) {
475 if (broken_) {
476 return false;
477 }
478 dataOk = false; // the answer is still read, to stay in step
479 }
480 } else if (!readLiteral(static_cast<size_t>(size), literal)) {
481 return false;
482 }
483 literalSize_ = size;
484 }
485 }
486 }

492 bool readHandled(size_t size, LiteralHandler& handler) {
493 if (maxLiteralSize_ > 0 && size > maxLiteralSize_) {
494 TinyLogger::instance().warn(String("IMAP: message of ") + String(static_cast<unsigned long>(size)) +
495 " bytes is over the limit of " +
496 String(static_cast<unsigned long>(maxLiteralSize_)) + ", skipped");
497 LiteralStream rest(*this, size);
498 while (rest.read() >= 0) {
499 }
500 if (rest.timedOut()) {
501 broken_ = true;
502 TinyLogger::instance().error("IMAP: timeout while skipping message data");
503 }
504 return false;
505 }
506 LiteralStream data(*this, size);
507 bool ok = handler.handle(data, size);
508 while (data.read() >= 0) {
509 }
510 if (data.timedOut()) {
511 broken_ = true;
512 TinyLogger::instance().error("IMAP: timeout while reading message data");
513 return false;
514 }
515 return ok;
516 }
517
519 bool readLiteral(size_t size, String* target) {
520 if (maxLiteralSize_ > 0 && size > maxLiteralSize_) {
521 broken_ = true; // the data is not read: the stream is out of step
522 TinyLogger::instance().error(String("IMAP: message of ") + String(static_cast<unsigned long>(size)) +
523 " bytes is over the limit of " + String(static_cast<unsigned long>(maxLiteralSize_)));
524 return false;
525 }
526 if (target != nullptr) {
527 *target = "";
528 target->reserve(size);
529 }
530 char buffer[64];
531 size_t left = size;
532 while (left > 0) {
533 size_t chunk = left < sizeof(buffer) ? left : sizeof(buffer);
534 size_t n = readBytes(buffer, chunk);
535 if (n != chunk) {
536 broken_ = true;
537 TinyLogger::instance().error("IMAP: timeout while reading message data");
538 return false;
539 }
540 if (target != nullptr) {
541 for (size_t i = 0; i < n; i++) {
542 *target += buffer[i];
543 }
544 }
545 left -= n;
546 }
547 return true;
548 }
